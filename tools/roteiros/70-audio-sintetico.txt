#   no Pi (ou aqui, com 127.0.0.1):  psd --synthetic --synthetic-audio --listen 127.0.0.1:7700
#   target\release\psc.exe 127.0.0.1:7700 --audio-dump .cache\audio-sintetico.wav --script tools\roteiros\70-audio-sintetico.txt
